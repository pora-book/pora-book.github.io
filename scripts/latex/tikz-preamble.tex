% Standalone preamble for pre-rendering the book's tikzpicture figures to SVG
% (LaTeXML can't render them). Mirrors the TikZ setup from the book's preamble.
\usepackage{amsmath,amssymb,amsfonts,bm,mathtools}

% Shared math macros — loaded BEFORE tikz on purpose: pora-macros defines
% \node, \edge and \graph (graph-theory shorthands) which would otherwise
% clobber TikZ's own \node/\edge/\graph commands and collapse every diagram.
% Loading tikz afterwards lets its definitions win.
\input{pora-macros.tex}
